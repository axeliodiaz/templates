<script setup>
import {ref,computed,onMounted,onBeforeUnmount,nextTick,watch} from 'vue'
import {animate,stagger} from 'motion'
const tab=ref('Home'),category=ref('All'),search=ref(''),range=ref('Week'),dark=ref(false),notice=ref(''),session=ref(null),seconds=ref(0),running=ref(false),goal=ref(10000),goalDraft=ref(10000),done=ref([]),root=ref(null)
const workouts=[{id:1,title:'Full-body mobility',category:'Mobility',minutes:15,level:'Beginner',icon:'↗'},{id:2,title:'Steady cardio',category:'Cardio',minutes:20,level:'Moderate',icon:'↻'},{id:3,title:'Strength basics',category:'Strength',minutes:25,level:'Moderate',icon:'+'},{id:4,title:'Evening yoga',category:'Yoga',minutes:12,level:'Beginner',icon:'☾'}]
const filtered=computed(()=>workouts.filter(w=>(category.value==='All'||w.category===category.value)&&w.title.toLowerCase().includes(search.value.toLowerCase())))
const bars=computed(()=>range.value==='Week'?[8432,9520,12100,14020,11300,8640,13240]:range.value==='Month'?[33000,41000,38000,52000]:range.value==='3 Months'?[112000,145000,168000]:[160000,180000,210000,190000,225000,204000,230000,240000,220000,245000,260000,250000])
let timer,media,animations=[],epoch=0,disposed=false
function stop(){clearInterval(timer);running.value=false}
function start(w){stop();session.value=w;seconds.value=0;notice.value='Local timer. No health data or device connected.'}
function toggle(){if(running.value)stop();else {running.value=true;timer=setInterval(()=>seconds.value++,1000)}}
function complete(){if(session.value&&!done.value.includes(session.value.id))done.value.push(session.value.id);stop();session.value=null;notice.value='Session marked complete in this demo only.'}
function clearMotion(){animations.forEach(a=>a.stop());animations=[];root.value?.querySelectorAll('.fit-card').forEach(e=>{e.style.opacity=1;e.style.transform='none'})}
async function enter(){const t=++epoch;clearMotion();await nextTick();if(disposed||t!==epoch||media?.matches||!root.value)return;animations.push(animate(root.value.querySelectorAll('.fit-card'),{opacity:[0,1],y:[6,0]},{duration:.25,delay:stagger(.04)}))}
watch(tab,()=>{stop();enter()});onMounted(()=>{media=matchMedia('(prefers-reduced-motion: reduce)');media.addEventListener('change',clearMotion);enter()});onBeforeUnmount(()=>{disposed=true;epoch++;stop();clearMotion();media?.removeEventListener('change',clearMotion)})
</script>

## Guide and credits

Standalone Vue3 layout with scoped tokens, local timer, category/search filters, range views, goal validation and demo completion state. Replace sample metrics through your own health-data integration and consent controls. No persistence or health API is bundled. Light/dark modes and reduced-motion behavior are included; timer/animations clean up on unmount.

Pattern reference: [Muiz](https://x.com/IamMuhammadMuiz/status/2108456553770520806). Original artwork-free implementation. No source photos, illustrations, personal metrics or branding copied.
